        ID          string

        CompletedAt time.Time

        return &Agent{
                ID:        id,
                Name:      name,
                Type:      agentType,
                Config:    make(map[string]interface{}),
                Status:    "idle",
                CreatedAt: time.Now(),
        }
}

func (a *Agent) SetConfig(key string, value interface{}) {
        a.Config[key] = value
}

func (a *Agent) GetConfig(key string) (interface{}, bool) {
        value, exists := a.Config[key]
        return value, exists
}

        return nil, fmt.Errorf("Execute not implemented for agent type: %s", a.Type)
}

// GetStatus returns the agent status
func (a *Agent) GetStatus() string {
        return a.Status
}

// GetDuration returns the execution duration
func (a *Agent) GetDuration() time.Duration {
        if a.StartedAt.IsZero() {
                return 0
        }
        if a.CompletedAt.IsZero() {
                return time.Since(a.StartedAt)
        }
        return a.CompletedAt.Sub(a.StartedAt)
}

// Registry manages available agent types
type Registry struct {
        agents map[string]func(id, name string) *Agent
        mu     sync.RWMutex
}

// NewRegistry creates a new agent registry
func NewRegistry() *Registry {
        return &Registry{
                agents: make(map[string]func(id, name string) *Agent),
        }
}

// Register registers an agent type
func (r *Registry) Register(agentType string, factory func(id, name string) *Agent) error {
        r.mu.Lock()
        defer r.mu.Unlock()

        if _, exists := r.agents[agentType]; exists {
                return fmt.Errorf("agent type %s already registered", agentType)
        }

        r.agents[agentType] = factory
        return nil
}

// Create creates an agent of the specified type
func (r *Registry) Create(agentType, id, name string) (*Agent, error) {
        r.mu.RLock()
        defer r.mu.RUnlock()

        factory, exists := r.agents[agentType]
        if !exists {
                return nil, fmt.Errorf("agent type %s not registered", agentType)
        }

        return factory(id, name), nil
}

// GetTypes returns all registered agent types
func (r *Registry) GetTypes() []string {
        r.mu.RLock()
        defer r.mu.RUnlock()

        types := make([]string, 0, len(r.agents))
        for agentType := range r.agents {
                types = append(types, agentType)
        }

        return types
}

// Execute deploys the application
func (a *DeployAgent) Execute(ctx context.Context, input interface{}) (interface{}, error) {
        a.Agent.Status = "running"
        a.Agent.StartedAt = time.Now()

        // Handle both single test result and multiple test results
        var allTestsPassed bool
        var testResults []*TestResult

        switch v := input.(type) {
        case *TestResult:
                testResults = []*TestResult{v}
                allTestsPassed = v.Success
        case []*TestResult:
                testResults = v
                allTestsPassed = true
                for _, tr := range testResults {
                        if !tr.Success {
                                allTestsPassed = false
                                break
                        }
                }
        case []interface{}:
                testResults = make([]*TestResult, 0)
                allTestsPassed = true
                for _, item := range v {
                        if tr, ok := item.(*TestResult); ok {
                                testResults = append(testResults, tr)
                                if !tr.Success {
                                        allTestsPassed = false
                                }
                        }
                }
        default:
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, fmt.Errorf("invalid input type, expected *TestResult or []*TestResult")
        }

        // Conditional deployment - only deploy if all tests pass
        if !allTestsPassed {
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, fmt.Errorf("deployment skipped: tests failed")
        }

        // Simulate deployment
        select {
        case <-time.After(100 * time.Millisecond):
        case <-ctx.Done():
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, ctx.Err()
        }

        result := &DeployResult{
                Success:     true,
                Environment: a.Environment,
                Version:     fmt.Sprintf("v1.0.%d", time.Now().Unix()),
                URL:         fmt.Sprintf("https://%s.example.com", a.Environment),
                DeployTime:  time.Since(a.Agent.StartedAt),
                DeployedAt:  time.Now(),
        }

        a.Agent.Status = "completed"
        a.Agent.CompletedAt = time.Now()
        return result, nil
}

// CICDPipeline creates a CI/CD pipeline workflow
func CICDPipeline() (*orchestrator.Orchestrator, error) {
        // Build DAG
        builder := orchestrator.NewWorkflowBuilder()

        // Add nodes
        builder.AddNode("build", "Build Application", "build")

        // Test nodes - run in parallel on different platforms
        builder.AddNode("test_linux", "Unit Tests (Linux)", "test")
        builder.AddNode("test_windows", "Unit Tests (Windows)", "test")
        builder.AddNode("test_macos", "Unit Tests (macOS)", "test")
        builder.AddNode("test_integration", "Integration Tests", "test")
        builder.AddNode("security_scan", "Security Scan", "security_scan")

        // Lint node
        builder.AddNode("lint", "Code Linting", "lint")

        // Deploy node
        builder.AddNode("deploy_staging", "Deploy to Staging", "deploy")

        // Add edges
        // Build -> all tests and security scan (parallel)
        builder.AddEdge("build", "test_linux", 1.0)
        builder.AddEdge("build", "test_windows", 1.0)
        builder.AddEdge("build", "test_macos", 1.0)
        builder.AddEdge("build", "test_integration", 1.0)
        builder.AddEdge("build", "security_scan", 1.0)

        // Build -> lint
        builder.AddEdge("build", "lint", 1.0)

        // All tests -> deploy (conditional)
        builder.AddEdge("test_linux", "deploy_staging", 1.0)
        builder.AddEdge("test_windows", "deploy_staging", 1.0)
        builder.AddEdge("test_macos", "deploy_staging", 1.0)
        builder.AddEdge("test_integration", "deploy_staging", 1.0)

        // Security scan -> deploy
        builder.AddEdge("security_scan", "deploy_staging", 1.0)

        // Lint -> deploy
        builder.AddEdge("lint", "deploy_staging", 1.0)

        // Validate
        if err := builder.Validate(); err != nil {
                return nil, err
        }

        dag := builder.Build()

        // Create orchestrator
        orc := orchestrator.NewOrchestrator(dag)

        // Register agents
        orc.RegisterAgent("build", NewBuildAgent("build-1", "Build Agent", "/app"))
        orc.RegisterAgent("test_linux", NewTestAgent("test-linux", "Linux Tests", "linux", "unit"))
        orc.RegisterAgent("test_windows", NewTestAgent("test-windows", "Windows Tests", "windows", "unit"))
        orc.RegisterAgent("test_macos", NewTestAgent("test-macos", "macOS Tests", "macos", "unit"))
        orc.RegisterAgent("test_integration", NewTestAgent("test-integration", "Integration Tests", "linux", "integration"))
        orc.RegisterAgent("security_scan", NewSecurityScanAgent("security-1", "Security Scanner", "snyk"))
        orc.RegisterAgent("lint", NewLintAgent("lint-1", "Linter", "golangci-lint"))
        orc.RegisterAgent("deploy_staging", NewDeployAgent("deploy-1", "Staging Deployer", "staging"))

        return orc, nil
}

// Step 2: Parse raw string logs into structured LogEntry structs
type LogParser struct{}

func (p *LogParser) Parse(rawLines []string) []LogEntry {
        fmt.Printf("[2. Parse] Parsing %d raw log entries...\n", len(rawLines))
        var entries []LogEntry

        for _, line := range rawLines {
                parts := strings.Split(line, " | ")
                if len(parts) != 6 {
                        continue // Skip malformed lines
                }

                t, err := time.Parse(time.RFC3339, strings.TrimSpace(parts[0]))
                if err != nil {
                        continue
                }

                status, err := strconv.Atoi(strings.TrimSpace(parts[3]))
                if err != nil {
                        continue
                }

                latency, err := strconv.Atoi(strings.TrimSpace(parts[4]))
                if err != nil {
                        continue
                }

                entries = append(entries, LogEntry{
                        Timestamp:  t,
                        Service:    strings.TrimSpace(parts[1]),
                        Level:      strings.TrimSpace(parts[2]),
                        StatusCode: status,
                        LatencyMs:  latency,
                        Message:    strings.TrimSpace(parts[5]),
                })
        }

        return entries
}

// Step 3: Analyze entries and calculate aggregate health metrics
type LogAnalyzer struct{}

func (a *LogAnalyzer) Analyze(entries []LogEntry) AnalysisReport {
        fmt.Println("[3. Analyze] Computing latency distribution and error breakdown...")

        var totalLatency int
        var latencies []int
        serviceErrors := make(map[string]int)
        totalErrors := 0

        for _, entry := range entries {
                totalLatency += entry.LatencyMs
                latencies = append(latencies, entry.LatencyMs)

                if entry.StatusCode >= 500 || entry.Level == "ERROR" {
                        totalErrors++
                        serviceErrors[entry.Service]++
                }
        }

        totalCount := len(entries)
        var avgLatency float64
        var p95Latency int
        var errorRate float64

        if totalCount > 0 {
                avgLatency = float64(totalLatency) / float64(totalCount)
                errorRate = (float64(totalErrors) / float64(totalCount)) * 100

                sort.Ints(latencies)
                p95Index := int(float64(totalCount) * 0.95)
                if p95Index >= totalCount {
                        p95Index = totalCount - 1
                }
                p95Latency = latencies[p95Index]
        }

        return AnalysisReport{
                TotalRequests: totalCount,
                TotalErrors:   totalErrors,
                ErrorRate:     errorRate,
                AvgLatencyMs:  avgLatency,
                P95LatencyMs:  p95Latency,
                ServiceErrors: serviceErrors,
                GeneratedAt:   time.Now(),
        }
}

        startTime := time.Now()
        fmt.Printf("\n>>> Starting Batch Job for date: %s <<<\n", batchDate)

        rawLogs, err := c.collector.CollectLogs(ctx, batchDate)
        if err != nil {
                return fmt.Errorf("collection failed: %w", err)
        }

        // Step 2: Parse

        metrics := c.analyzer.Analyze(entries)

        // Step 4: Generate Report
        reportContent := c.reporter.Generate(metrics)

        if err := c.mailer.SendReport(recipients, subject, reportContent); err != nil {
                return fmt.Errorf("notification failed: %w", err)
        }

        fmt.Printf(">>> Batch Job Finished in %v <<<\n\n", time.Since(startTime).Round(time.Millisecond))
        return nil
}

// ScheduleCron simulates a cron scheduler running at periodic intervals or specific times
func (c *NightlyJobCoordinator) ScheduleCron(ctx context.Context, interval time.Duration, recipients []string) {
        ticker := time.NewTicker(interval)
        defer ticker.Stop()

        for {
                select {
                case <-ctx.Done():
                        fmt.Println("[Scheduler] Stopped batch schedule listener.")
                        return
                case t := <-ticker.C:
                        dateStr := t.Format("2006-01-02")
                        _ = c.Run(ctx, dateStr, recipients)
                }
        }
}

// DocumentProcessor handles OCR extraction and validation
type DocumentProcessor struct{}

func (dp *DocumentProcessor) Process(doc *Document) error {
        if doc.RawPayload == "" {
                return errors.New("document body is empty")
        }

        // Simulated OCR / Key-Value Extraction
        doc.ExtractedData = make(map[string]string)
        lines := strings.Split(doc.RawPayload, ";")
        for _, line := range lines {
                parts := strings.SplitN(line, "=", 2)
                if len(parts) == 2 {
                        key := strings.TrimSpace(parts[0])
                        val := strings.TrimSpace(parts[1])
                        doc.ExtractedData[key] = val
                }
        }

        // Validate presence of essential invoice metadata
        if _, ok := doc.ExtractedData["TAX_ID"]; !ok {
                return errors.New("OCR validation failed: TAX_ID missing from document")
        }
        if _, ok := doc.ExtractedData["INVOICE_TOTAL"]; !ok {
                return errors.New("OCR validation failed: INVOICE_TOTAL missing from document")
        }

        doc.Validated = true
        return nil
}

// Step 1: Ingest and process documents
func (ae *ApprovalEngine) IngestAndProcess(req *ExpenseRequest) error {
        fmt.Printf("[Step 1] Ingesting Request %s and processing document: %s\n", req.RequestID, req.Document.FileName)

        if err := ae.docProcessor.Process(&req.Document); err != nil {
                req.State = StateRejected
                req.AddAudit("SYSTEM", "OCR_ENGINE", false, err.Error())
                ae.notifier.Notify(req.SubmitterID, "Submission Rejected", "Invalid document: "+err.Error())
                return err
        }

        req.State = StateDocProcessed
        fmt.Println("  Document processed successfully. Extracted fields:")
        for k, v := range req.Document.ExtractedData {
                fmt.Printf("   - %s: %s\n", k, v)
        }

        // Route to first approval stage
        req.State = StatePendingManager
        ae.notifier.Notify("manager@company.com", "Action Required", fmt.Sprintf("Request %s awaiting manager review", req.RequestID))
        return nil
}

// Step 2: Line Manager Approval
func (ae *ApprovalEngine) ReviewByManager(req *ExpenseRequest, approver string, approve bool, comment string) error {
        if req.State != StatePendingManager {
                return fmt.Errorf("invalid transition: request is not in PENDING_MANAGER (current: %s)", req.State)
        }

        req.AddAudit("MANAGER", approver, approve, comment)

        if !approve {
                req.State = StateRejected
                ae.notifier.Notify(req.SubmitterID, "Request Rejected", "Manager rejected request: "+comment)
                return nil
        }

        req.State = StatePendingFinance
        ae.notifier.Notify("finance@company.com", "Action Required", fmt.Sprintf("Request %s passed manager, needs finance audit", req.RequestID))
        return nil
}

// Step 3: Finance Audit & Policy Enforcement
func (ae *ApprovalEngine) ReviewByFinance(req *ExpenseRequest, auditor string, approve bool, comment string) error {
        if req.State != StatePendingFinance {
                return fmt.Errorf("invalid transition: request is not in PENDING_FINANCE (current: %s)", req.State)
        }

        req.AddAudit("FINANCE", auditor, approve, comment)

        if !approve {
                req.State = StateRejected
                ae.notifier.Notify(req.SubmitterID, "Request Rejected", "Finance rejected request: "+comment)
                return nil
        }

        // Policy Rule: Expenses over $5,000 escalate to Executive Approval
        if req.Amount > 5000.00 {
                req.State = StatePendingExecutive
                ae.notifier.Notify("vp_finance@company.com", "High-Value Approval Required", fmt.Sprintf("Request %s ($%.2f) requires executive clearance", req.RequestID, req.Amount))
        } else {
                req.State = StateApproved
                ae.completeWorkflow(req)
        }

        return nil
}

// Step 4: Final Executive Approval (for high-value purchases)
func (ae *ApprovalEngine) ReviewByExecutive(req *ExpenseRequest, exec string, approve bool, comment string) error {
        if req.State != StatePendingExecutive {
                return fmt.Errorf("invalid transition: request is not in PENDING_EXECUTIVE (current: %s)", req.State)
        }

        req.AddAudit("EXECUTIVE", exec, approve, comment)

        if !approve {
                req.State = StateRejected
                ae.notifier.Notify(req.SubmitterID, "Request Rejected", "Executive rejected request: "+comment)
                return nil
        }

        req.State = StateApproved
        ae.completeWorkflow(req)
        return nil
}

// Step 5: Finalization & Notification
func (ae *ApprovalEngine) completeWorkflow(req *ExpenseRequest) {
        req.State = StateCompleted
        fmt.Printf("[Finalize] Request %s approved. Initiating payment disbursement.\n", req.RequestID)
        ae.notifier.Notify(req.SubmitterID, "Request Approved", fmt.Sprintf("Your request %s for $%.2f has been fully approved.", req.RequestID, req.Amount))
        ae.notifier.Notify("erp_sync@internal", "ERP Ingestion", fmt.Sprintf("Syncing request %s into Ledger", req.RequestID))
}

type OrderFulfillmentOrchestrator struct {
        user      UserService
        inventory InventoryService
        payment   PaymentService
        notify    NotificationService
}

func NewOrderFulfillmentOrchestrator(
        u UserService,
        inv InventoryService,
        p PaymentService,
        n NotificationService,
) *OrderFulfillmentOrchestrator {
        return &OrderFulfillmentOrchestrator{
                user:      u,
                inventory: inv,
                payment:   p,
                notify:    n,
        }
}

func (o *OrderFulfillmentOrchestrator) Execute(ctx context.Context, order *OrderContext) error {
        var compensations []CompensationStep

        // Rollback unrolls successfully executed mutations in LIFO order
        rollback := func() {
                if len(compensations) == 0 {
                        return
                }
                fmt.Println("[Orchestrator] Step failed. Executing compensations...")
                for i := len(compensations) - 1; i >= 0; i-- {
                        // Detach cleanup timeout from parent context cancellation
                        cleanCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
                        if err := compensations[i](cleanCtx); err != nil {
                                fmt.Printf("  [Alert] Rollback step failed: %v\n", err)
                        }
                        cancel()
                }
        }

        // -------------------------------------------------------------------------
        // Step 1: Validate User
        // -------------------------------------------------------------------------
        fmt.Printf("[Step 1] Validating user: %s...\n", order.UserID)
        userProfile, err := o.user.ValidateUser(ctx, order.UserID)
        if err != nil {
                return fmt.Errorf("user validation failed: %w", err)
        }
        order.User = userProfile
        fmt.Printf("  User %s valid (Email: %s)\n", userProfile.UserID, userProfile.Email)

        // -------------------------------------------------------------------------
        // Step 2: Check & Hold Inventory (Depends on valid user)
        // -------------------------------------------------------------------------
        fmt.Printf("[Step 2] Checking stock & holding %d unit(s) of %s...\n", order.Quantity, order.ItemID)
        resID, err := o.inventory.CheckAndHold(ctx, order.ItemID, order.Quantity)
        if err != nil {
                return fmt.Errorf("inventory check failed: %w", err)
        }
        order.ReservationID = resID
        fmt.Printf("  Stock reserved. ReservationID: %s\n", resID)

        // Register compensation to release hold if downstream steps fail
        compensations = append(compensations, func(c context.Context) error {
                return o.inventory.ReleaseHold(c, resID)
        })

        // -------------------------------------------------------------------------
        // Step 3: Process Payment (Depends on User + ReservationID + Amount)
        // -------------------------------------------------------------------------
        fmt.Printf("[Step 3] Processing charge of $%.2f for user %s...\n", order.Amount, order.User.UserID)
        txnID, err := o.payment.ProcessPayment(ctx, order.User.UserID, order.Amount, order.ReservationID)
        if err != nil {
                rollback()
                return fmt.Errorf("payment processing failed: %w", err)
        }
        order.PaymentTxnID = txnID
        fmt.Printf("  Payment authorized. TransactionID: %s\n", txnID)

        // Register compensation to refund payment if subsequent steps fail
        compensations = append(compensations, func(c context.Context) error {
                return o.payment.RefundPayment(c, txnID)
        })

        // -------------------------------------------------------------------------
        // Step 4: Send Confirmation (Depends on PaymentTxnID + User Email)
        // -------------------------------------------------------------------------
        fmt.Printf("[Step 4] Sending confirmation receipt to %s...\n", order.User.Email)
        if err := o.notify.SendConfirmation(ctx, order.User.Email, order.OrderID, order.PaymentTxnID); err != nil {
                rollback()
                return fmt.Errorf("notification failed: %w", err)
        }
        order.ConfirmationSent = true
        fmt.Println("  Confirmation sent successfully.")

        // -------------------------------------------------------------------------
        // Step 5: Update/Commit Inventory (Depends on ReservationID + PaymentTxnID)
        // -------------------------------------------------------------------------
        fmt.Printf("[Step 5] Finalizing stock deduction for reservation %s...\n", order.ReservationID)
        if err := o.inventory.CommitStock(ctx, order.ReservationID, order.PaymentTxnID); err != nil {
                rollback()
                return fmt.Errorf("inventory final commit failed: %w", err)
        }
        order.StockCommitted = true
        fmt.Println("  Inventory ledger updated permanently.")

        return nil
}

// -----------------------------------------------------------------------------
// 1. Domain Event Models
// -----------------------------------------------------------------------------

type RawEvent struct {
        ID        string
        UserID    string
        Amount    float64
        Type      string // "TRANSACTION", "PING", "REFUND"
        Timestamp time.Time
}

type EnrichedEvent struct {
        RawEvent
        Country   string
        Tier      string
        RiskScore int // 0 (safe) to 100 (critical risk)
}

type AggregatedRecord struct {
        Event          EnrichedEvent
        WindowSpend    float64 // Cumulative spend in sliding window
        WindowCount    int     // Transaction count in sliding window
        WindowDuration time.Duration
}

// -----------------------------------------------------------------------------
// 2. Stage 1: Ingest
// -----------------------------------------------------------------------------

func Ingest(ctx context.Context, events []RawEvent) <-chan RawEvent {
        out := make(chan RawEvent)
        go func() {
                defer close(out)
                for _, evt := range events {
                        select {
                        case <-ctx.Done():
                                return
                        case out <- evt:
                                fmt.Printf("[1. Ingest] Event %s | User: %s | Amount: $%.2f | Type: %s\n",
                                        evt.ID, evt.UserID, evt.Amount, evt.Type)
                                time.Sleep(30 * time.Millisecond) // Simulated stream ingestion delay
                        }
                }
        }()
        return out
}

// -----------------------------------------------------------------------------
// 3. Stage 2: Filter
// -----------------------------------------------------------------------------

// Filter drops non-actionable heartbeats and malformed zero/negative amounts
func Filter(ctx context.Context, in <-chan RawEvent) <-chan RawEvent {
        out := make(chan RawEvent)
        go func() {
                defer close(out)
                for evt := range in {
                        if evt.Type == "PING" || evt.Amount <= 0 {
                                fmt.Printf("  [2. Filter] Dropping noise/invalid event: %s (%s, $%.2f)\n",
                                        evt.ID, evt.Type, evt.Amount)
                                continue
                        }
                        select {
                        case <-ctx.Done():
                                return
                        case out <- evt:
                        }
                }
        }()
        return out
}

// -----------------------------------------------------------------------------
// 4. Stage 3: Enrich
// -----------------------------------------------------------------------------

type UserProfile struct {
        Country   string
        Tier      string
        RiskScore int
}

// Enrich decorates raw events with user tier, country, and risk profile
func Enrich(ctx context.Context, in <-chan RawEvent) <-chan EnrichedEvent {
        out := make(chan EnrichedEvent)

        // Simulated fast in-memory user cache
        cache := map[string]UserProfile{
                "usr_101": {Country: "US", Tier: "VIP", RiskScore: 12},
                "usr_102": {Country: "DE", Tier: "Standard", RiskScore: 35},
                "usr_103": {Country: "NG", Tier: "Standard", RiskScore: 88},
        }

        go func() {
                defer close(out)
                for evt := range in {
                        profile, exists := cache[evt.UserID]
                        if !exists {
                                profile = UserProfile{Country: "Unknown", Tier: "Standard", RiskScore: 50}
                        }

                        enriched := EnrichedEvent{
                                RawEvent:  evt,
                                Country:   profile.Country,
                                Tier:      profile.Tier,
                                RiskScore: profile.RiskScore,
                        }

                        fmt.Printf("  [3. Enrich] Event %s -> Country: %s | Tier: %s | Risk: %d\n",
                                enriched.ID, enriched.Country, enriched.Tier, enriched.RiskScore)

                        select {
                        case <-ctx.Done():
                                return
                        case out <- enriched:
                        }
                }
        }()
        return out
}

// -----------------------------------------------------------------------------
// 5. Stage 4: Aggregate (Sliding Window Velocity)
// -----------------------------------------------------------------------------

type windowItem struct {
        amount    float64
        timestamp time.Time
}

// Aggregate tracks rolling spend and event frequency per user across windowDuration
func Aggregate(ctx context.Context, windowDuration time.Duration, in <-chan EnrichedEvent) <-chan AggregatedRecord {
        out := make(chan AggregatedRecord)
        windows := make(map[string][]windowItem)

        go func() {
                defer close(out)
                for evt := range in {
                        now := evt.Timestamp
                        cutoff := now.Add(-windowDuration)

                        // Evict expired entries outside the sliding time frame
                        var active []windowItem
                        var windowSpend float64
                        for _, item := range windows[evt.UserID] {
                                if item.timestamp.After(cutoff) {
                                        active = append(active, item)
                                        windowSpend += item.amount
                                }
                        }

                        // Add current event to state
                        active = append(active, windowItem{amount: evt.Amount, timestamp: now})
                        windowSpend += evt.Amount
                        windows[evt.UserID] = active

                        record := AggregatedRecord{
                                Event:          evt,
                                WindowSpend:    windowSpend,
                                WindowCount:    len(active),
                                WindowDuration: windowDuration,
                        }

                        fmt.Printf("  [4. Aggregate] User %s: Rolling %v spend = $%.2f (%d txns)\n",
                                evt.UserID, windowDuration, record.WindowSpend, record.WindowCount)

                        select {
                        case <-ctx.Done():
                                return
                        case out <- record:
                        }
                }
        }()
        return out
}

// -----------------------------------------------------------------------------
// 6. Stages 5 & 6: Store and Alert
// -----------------------------------------------------------------------------

type StorageSink struct {
        mu      sync.Mutex
        records []AggregatedRecord
}

func (s *StorageSink) Save(rec AggregatedRecord) {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.records = append(s.records, rec)
        fmt.Printf("  [5. Store] Saved %s to persistence sink (Total stored: %d)\n",
                rec.Event.ID, len(s.records))
}

// StoreAndAlert persists records and dispatches alerts on anomaly triggers
func StoreAndAlert(in <-chan AggregatedRecord, sink *StorageSink) {
        for rec := range in {
                // Stage 5: Store
                sink.Save(rec)

                // Stage 6: Alert Rules
                // Rule A: Velocity spike — user spend exceeds $4,000 within the rolling window
                if rec.WindowSpend > 4000.00 {
                        fmt.Printf("  [6. ALERT ⚠️] Velocity Spike! User %s spent $%.2f within %v (Threshold: $4,000.00)\n",
                                rec.Event.UserID, rec.WindowSpend, rec.WindowDuration)
                }

                // Rule B: High-Risk outlier — RiskScore > 80 with single transaction > $1,000
                if rec.Event.RiskScore > 80 && rec.Event.Amount > 1000.00 {
                        fmt.Printf("  [6. ALERT 🚨] Fraud Risk! User %s (RiskScore: %d, Origin: %s) charged $%.2f\n",
                                rec.Event.UserID, rec.Event.RiskScore, rec.Event.Country, rec.Event.Amount)
                }
        }
}

// -----------------------------------------------------------------------------

func main() {
        ctx, cancel := context.WithCancel(context.Background())
        defer cancel()

        now := time.Now()

        // Simulated live stream containing valid transactions, noise, and suspicious bursts
        streamInput := []RawEvent{
                {ID: "evt_001", UserID: "usr_101", Amount: 250.00, Type: "TRANSACTION", Timestamp: now},
                {ID: "evt_002", UserID: "usr_102", Amount: 0.00, Type: "PING", Timestamp: now.Add(1 * time.Second)},
                {ID: "evt_003", UserID: "usr_102", Amount: 850.00, Type: "TRANSACTION", Timestamp: now.Add(2 * time.Second)},
                {ID: "evt_004", UserID: "usr_103", Amount: 1400.00, Type: "TRANSACTION", Timestamp: now.Add(3 * time.Second)},
                {ID: "evt_005", UserID: "usr_101", Amount: 4200.00, Type: "TRANSACTION", Timestamp: now.Add(4 * time.Second)},
        }

        storage := &StorageSink{}
        slidingWindow := 10 * time.Second

        fmt.Println("=== Starting Real-Time Event Processing Pipeline ===")

        // Connect pipeline stages
        rawStream := Ingest(ctx, streamInput)
        filteredStream := Filter(ctx, rawStream)
        enrichedStream := Enrich(ctx, filteredStream)

}

// Helper to safely extract and clone map data between nodes
func extractMap(input interface{}) map[string]interface{} {
        if m, ok := input.(map[string]interface{}); ok {
                res := make(map[string]interface{}, len(m)+8)
                for k, v := range m {
                        res[k] = v
                }
                return res
        }
        if slice, ok := input.([]interface{}); ok {
                merged := make(map[string]interface{})
                for _, item := range slice {
                        if m, ok := item.(map[string]interface{}); ok {
                                for k, v := range m {
                                        merged[k] = v
                                }
                        }
                }
                return merged
        }
        return make(map[string]interface{})
}

type NominatimResult struct {
        DisplayName string `json:"display_name"`
        Name        string `json:"name"`
        Type        string `json:"type"`
        Lat         string `json:"lat"`
        Lon         string `json:"lon"`
}

func (a *AirportResolverAgent) Execute(ctx context.Context, input interface{}) (interface{}, error) {
        data := extractMap(input)

        originCity, _ := data["origin_city"].(string)
        destCity, _ := data["dest_city"].(string)

        if originCity == "" || destCity == "" {
                return nil, fmt.Errorf("origin_city and dest_city are required")
        }

        // Dynamically query external service for origin airport
        origAirport, origCode, err := a.lookupAirport(ctx, originCity)
        if err != nil {
                return nil, fmt.Errorf("failed to resolve origin airport for %s: %w", originCity, err)
        }

        // Dynamically query external service for destination airport
        destAirport, destCode, err := a.lookupAirport(ctx, destCity)
        if err != nil {
                return nil, fmt.Errorf("failed to resolve destination airport for %s: %w", destCity, err)
        }

        data["origin_airport"] = origAirport
        data["origin_code"] = origCode
        data["dest_airport"] = destAirport
        data["dest_code"] = destCode
        data["resolved_by"] = a.Name

        return data, nil
}

func (a *AirportResolverAgent) lookupAirport(ctx context.Context, city string) (airportName string, iata string, err error) {
        // Query OpenStreetMap Nominatim for the major airport of the city
        query := fmt.Sprintf("%s international airport", city)
        reqURL := fmt.Sprintf("https://nominatim.openstreetmap.org/search?q=%s&format=json&limit=1", url.QueryEscape(query))

        req, err := http.NewRequestWithContext(ctx, http.MethodGet, reqURL, nil)
        if err != nil {
                return "", "", err
        }
        req.Header.Set("User-Agent", "MultiAgentGraphWorkflow/1.0 (travel-agent)")

        resp, err := a.Client.Do(req)
        if err != nil {
                return "", "", err
        }
        defer resp.Body.Close()

        body, err := io.ReadAll(resp.Body)
        if err != nil {
                return "", "", err
        }

        var results []NominatimResult
        if err := json.Unmarshal(body, &results); err != nil || len(results) == 0 {
                // Fallback to basic derivation if no response
                return fmt.Sprintf("%s International Airport", city), strings.ToUpper(city[:3]), nil
        }

        fullName := results[0].DisplayName
        parts := strings.Split(fullName, ",")
        resolvedName := strings.TrimSpace(parts[0])

        // Approximate IATA from city name or recognized code
        code := strings.ToUpper(city[:3])
        return fmt.Sprintf("%s (%s)", code, resolvedName), code, nil
}

        data["price"] = offer.Price

        data["searched_by"] = a.Name

func (a *FlightSearchAgent) searchLiveFlights(ctx context.Context, origin, dest, date string) (*FlightOffer, error) {
        // If APIBaseURL is provided, perform live HTTP flight search
        if a.APIBaseURL != "" {
                endpoint := fmt.Sprintf("%s/v2/shopping/flight-offers?originLocationCode=%s&destinationLocationCode=%s&departureDate=%s&adults=1",
                        a.APIBaseURL, origin, dest, date)

                req, err := http.NewRequestWithContext(ctx, http.MethodGet, endpoint, nil)
                if err != nil {
                        return nil, err
                }
                if a.APIKey != "" {
                        req.Header.Set("Authorization", "Bearer "+a.APIKey)
                }

                resp, err := a.Client.Do(req)
                if err != nil {
                        return nil, err
                }
                defer resp.Body.Close()

                if resp.StatusCode == http.StatusOK {
                        var result struct {
                                Data []struct {
                                        Price struct {
                                                Total    string `json:"total"`
                                                Currency string `json:"currency"`
                                        } `json:"price"`
                                        Itineraries []struct {
                                                Segments []struct {
                                                        CarrierCode string `json:"carrierCode"`
                                                        Number      string `json:"number"`
                                                        Departure   struct {
                                                                At string `json:"at"`
                                                        } `json:"departure"`
                                                        Arrival struct {
                                                                At string `json:"at"`
                                                        } `json:"arrival"`
                                                } `json:"segments"`
                                        } `json:"itineraries"`
                                } `json:"data"`
                        }
                        if err := json.NewDecoder(resp.Body).Decode(&result); err == nil && len(result.Data) > 0 {
                                first := result.Data[0]
                                seg := first.Itineraries[0].Segments[0]
                                return &FlightOffer{
                                        FlightNumber:  fmt.Sprintf("%s%s", seg.CarrierCode, seg.Number),
                                        Airline:       seg.CarrierCode,
                                        DepartureTime: seg.Departure.At,
                                        ArrivalTime:   seg.Arrival.At,
                                        Price:         fmt.Sprintf("%s %s", first.Price.Currency, first.Price.Total),
                                        Layovers:      fmt.Sprintf("%d stops", len(first.Itineraries[0].Segments)-1),
                                }, nil
                        }
                }
        }

        return nil, fmt.Errorf("no live flight provider connected")
}

// Execute trains the model
func (a *ModelTrainer) Execute(ctx context.Context, input interface{}) (interface{}, error) {
        a.Agent.Status = "running"
        a.Agent.StartedAt = time.Now()

        // Get feature extraction result
        _, ok := input.(*FeatureExtractionResult)
        if !ok {
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, fmt.Errorf("invalid input type, expected *FeatureExtractionResult")
        }

        // Simulate training based on hyperparameters
        var trainingDuration time.Duration
        epochs, _ := a.Hyperparams["epochs"].(int)
        batchSize, _ := a.Hyperparams["batch_size"].(int)
        learningRate, _ := a.Hyperparams["learning_rate"].(float64)

        // Training duration depends on hyperparameters
        durationMultiplier := float64(epochs) / float64(batchSize) * (1.0 / learningRate)
        trainingDuration = time.Duration(durationMultiplier * 50 * float64(time.Millisecond))

        select {
        case <-time.After(trainingDuration):
        case <-ctx.Done():
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, ctx.Err()
        }

        // Simulate different accuracies based on hyperparameters
        // Better hyperparameters = better accuracy
        baseAccuracy := 0.85
        accuracyBonus := (float64(epochs) / 100.0) * 0.05
        accuracyBonus += (1.0 / learningRate) * 0.02
        accuracyBonus += (1.0 / float64(batchSize)) * 0.01
        accuracy := math.Min(baseAccuracy+accuracyBonus, 0.98)

        precision := accuracy - 0.02
        recall := accuracy - 0.03
        f1Score := 2 * (precision * recall) / (precision + recall)

        result := &ModelTrainingResult{
                ModelID:      fmt.Sprintf("%s-%d", a.ModelType, time.Now().UnixNano()),
                ModelType:    a.ModelType,
                Hyperparams:  a.Hyperparams,
                Accuracy:     accuracy,
                Precision:    precision,
                Recall:       recall,
                F1Score:      f1Score,
                TrainingTime: time.Since(a.Agent.StartedAt),
                TrainedAt:    time.Now(),
        }

        a.Agent.Status = "completed"
        a.Agent.CompletedAt = time.Now()
        return result, nil
}

// ModelComparer compares multiple models and selects the best
type ModelComparer struct {
        *agent.Agent
        Metric string // "accuracy", "f1_score", etc.
}

// NewModelComparer creates a new model comparer agent
func NewModelComparer(id, name, metric string) *ModelComparer {
        return &ModelComparer{
                Agent:  agent.NewAgent(id, name, "model_comparer"),
                Metric: metric,
        }
}

// Execute compares models and selects the best
func (a *ModelComparer) Execute(ctx context.Context, input interface{}) (interface{}, error) {
        a.Agent.Status = "running"
        a.Agent.StartedAt = time.Now()

        // Handle both single and multiple model results
        var modelResults []*ModelTrainingResult

        switch v := input.(type) {
        case *ModelTrainingResult:
                modelResults = []*ModelTrainingResult{v}
        case []*ModelTrainingResult:
                modelResults = v
        case []interface{}:
                modelResults = make([]*ModelTrainingResult, 0)
                for _, item := range v {
                        if mr, ok := item.(*ModelTrainingResult); ok {
                                modelResults = append(modelResults, mr)
                        }
                }
        default:
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, fmt.Errorf("invalid input type, expected *ModelTrainingResult or []*ModelTrainingResult")
        }

        // Simulate comparison
        select {
        case <-time.After(40 * time.Millisecond):
        case <-ctx.Done():
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, ctx.Err()
        }

        // Find best model based on metric
        if len(modelResults) == 0 {
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, fmt.Errorf("no models to compare")
        }

        bestModel := modelResults[0]
        bestValue := a.getMetricValue(bestModel)

        for _, model := range modelResults[1:] {
                value := a.getMetricValue(model)
                if value > bestValue {
                        bestModel = model
                        bestValue = value
                }
        }

        result := &ModelComparisonResult{
                BestModelID:  bestModel.ModelID,
                BestAccuracy: bestModel.Accuracy,

// Execute deploys the model
func (a *ModelDeployer) Execute(ctx context.Context, input interface{}) (interface{}, error) {
        a.Agent.Status = "running"
        a.Agent.StartedAt = time.Now()

        // Get model comparison result
        comparisonResult, ok := input.(*ModelComparisonResult)
        if !ok {
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, fmt.Errorf("invalid input type, expected *ModelComparisonResult")
        }

        // Simulate deployment
        select {
        case <-time.After(120 * time.Millisecond):
        case <-ctx.Done():
                a.Agent.Status = "failed"
                a.Agent.CompletedAt = time.Now()
                return nil, ctx.Err()
        }

        result := &ModelDeploymentResult{
                Success:     true,
                ModelID:     comparisonResult.BestModelID,
                Environment: a.Environment,
                Endpoint:    fmt.Sprintf("https://api.%s.example.com/v1/predict", a.Environment),
                Version:     fmt.Sprintf("v1.0.%d", time.Now().Unix()),
                DeployTime:  time.Since(a.Agent.StartedAt),
                DeployedAt:  time.Now(),
        }

        a.Agent.Status = "completed"
        a.Agent.CompletedAt = time.Now()
        return result, nil
}

import (
        "errors"
        "fmt"
        "sync"
)

// DAG represents a Directed Acyclic Graph
type DAG struct {
        Nodes map[string]*Node
        Edges []*Edge
        mu    sync.RWMutex
}

// NewDAG creates a new DAG
func NewDAG() *DAG {
        return &DAG{
                Nodes: make(map[string]*Node),
                Edges: make([]*Edge, 0),
        }
}

// AddNode adds a node to the DAG
func (d *DAG) AddNode(node *Node) error {
        d.mu.Lock()
        defer d.mu.Unlock()

        if _, exists := d.Nodes[node.ID]; exists {
                return fmt.Errorf("node with ID %s already exists", node.ID)
        }

        d.Nodes[node.ID] = node
        return nil
}

// AddEdge adds an edge to the DAG
func (d *DAG) AddEdge(from, to string, weight float64) error {
        d.mu.Lock()
        defer d.mu.Unlock()

        if _, exists := d.Nodes[from]; !exists {
                return fmt.Errorf("node %s does not exist", from)
        }

        if _, exists := d.Nodes[to]; !exists {
                return fmt.Errorf("node %s does not exist", to)
        }

        // Check if edge already exists
        for _, edge := range d.Edges {
                if edge.From == from && edge.To == to {
                        return fmt.Errorf("edge from %s to %s already exists", from, to)
                }
        }

        // Add edge
        d.Edges = append(d.Edges, NewEdge(from, to, weight))

        // Update node inputs/outputs
        d.Nodes[from].Outputs = append(d.Nodes[from].Outputs, to)
        d.Nodes[to].Inputs = append(d.Nodes[to].Inputs, from)

        // Check for cycles
        if d.hasCycle() {
                // Rollback edge addition
                d.Edges = d.Edges[:len(d.Edges)-1]
                d.Nodes[from].Outputs = d.Nodes[from].Outputs[:len(d.Nodes[from].Outputs)-1]
                d.Nodes[to].Inputs = d.Nodes[to].Inputs[:len(d.Nodes[to].Inputs)-1]
                return errors.New("adding this edge would create a cycle")
        }

        return nil
}

// GetNode returns a node by ID
func (d *DAG) GetNode(id string) (*Node, error) {
        d.mu.RLock()
        defer d.mu.RUnlock()

        node, exists := d.Nodes[id]
        if !exists {
                return nil, fmt.Errorf("node %s does not exist", id)
        }

        return node, nil
}

// GetNodes returns all nodes
func (d *DAG) GetNodes() []*Node {
        d.mu.RLock()
        defer d.mu.RUnlock()

        nodes := make([]*Node, 0, len(d.Nodes))
        for _, node := range d.Nodes {
                nodes = append(nodes, node)
        }

        return nodes
}

// GetEdges returns all edges
func (d *DAG) GetEdges() []*Edge {
        d.mu.RLock()
        defer d.mu.RUnlock()

        edges := make([]*Edge, len(d.Edges))
        copy(edges, d.Edges)

        return edges
}

// GetRootNodes returns nodes with no inputs (root nodes)
func (d *DAG) GetRootNodes() []*Node {
        d.mu.RLock()
        defer d.mu.RUnlock()

        roots := make([]*Node, 0)
        for _, node := range d.Nodes {
                if len(node.Inputs) == 0 {
                        roots = append(roots, node)
                }
        }

        return roots
}

// GetLeafNodes returns nodes with no outputs (leaf nodes)
func (d *DAG) GetLeafNodes() []*Node {
        d.mu.RLock()
        defer d.mu.RUnlock()

        leaves := make([]*Node, 0)
        for _, node := range d.Nodes {
                if len(node.Outputs) == 0 {
                        leaves = append(leaves, node)
                }
        }

        return leaves
}

// GetDependents returns nodes that depend on the given node
func (d *DAG) GetDependents(nodeID string) []*Node {
        d.mu.RLock()
        defer d.mu.RUnlock()

        dependents := make([]*Node, 0)
        for _, edge := range d.Edges {
                if edge.From == nodeID {
                        if node, exists := d.Nodes[edge.To]; exists {
                                dependents = append(dependents, node)
                        }
                }
        }

        return dependents
}

// GetDependencies returns nodes that the given node depends on
func (d *DAG) GetDependencies(nodeID string) []*Node {
        d.mu.RLock()
        defer d.mu.RUnlock()

        dependencies := make([]*Node, 0)
        for _, edge := range d.Edges {
                if edge.To == nodeID {
                        if node, exists := d.Nodes[edge.From]; exists {
                                dependencies = append(dependencies, node)
                        }
                }
        }

        return dependencies
}

// hasCycle checks if the DAG has a cycle using DFS
func (d *DAG) hasCycle() bool {
        visited := make(map[string]bool)
        recursionStack := make(map[string]bool)

        for nodeID := range d.Nodes {
                if !visited[nodeID] {
                        if d.dfsCycleCheck(nodeID, visited, recursionStack) {
                                return true
                        }
                }
        }

        return false
}

// dfsCycleCheck performs DFS to detect cycles
func (d *DAG) dfsCycleCheck(nodeID string, visited, recursionStack map[string]bool) bool {
        visited[nodeID] = true
        recursionStack[nodeID] = true

        for _, edge := range d.Edges {
                if edge.From == nodeID {
                        if !visited[edge.To] {
                                if d.dfsCycleCheck(edge.To, visited, recursionStack) {
                                        return true
                                }
                        } else if recursionStack[edge.To] {
                                return true
                        }
                }
        }

        recursionStack[nodeID] = false
        return false
}

// TopologicalSort returns nodes in topological order
func (d *DAG) TopologicalSort() ([]*Node, error) {
        d.mu.RLock()
        defer d.mu.RUnlock()

        if d.hasCycle() {
                return nil, errors.New("cannot perform topological sort on graph with cycles")
        }

        // Kahn's algorithm
        inDegree := make(map[string]int)
        for nodeID := range d.Nodes {
                inDegree[nodeID] = 0
        }

        for _, edge := range d.Edges {
                inDegree[edge.To]++
        }

        queue := make([]*Node, 0)
        for nodeID, degree := range inDegree {
                if degree == 0 {
                        queue = append(queue, d.Nodes[nodeID])
                }
        }

        result := make([]*Node, 0)
        for len(queue) > 0 {
                node := queue[0]
                queue = queue[1:]
                result = append(result, node)

                for _, edge := range d.Edges {
                        if edge.From == node.ID {
                                inDegree[edge.To]--
                                if inDegree[edge.To] == 0 {
                                        queue = append(queue, d.Nodes[edge.To])
                                }
                        }
                }
        }

        if len(result) != len(d.Nodes) {
                return nil, errors.New("topological sort failed - graph has cycles")
        }

        return result, nil
}

        if len(nodes) == 0 {
                return fmt.Errorf("workflow has no nodes")
        }

        if err != nil {
                return fmt.Errorf("workflow validation failed: %w", err)
        }

        return nil
}

package orchestrator

import (
        "context"
        "fmt"
        "log"
        "sync"
        "time"

        "multi-agent-graph/agent"
        "multi-agent-graph/graph"
)

// Orchestrator manages workflow execution
type Orchestrator struct {
        DAG      *graph.DAG
        Agents   map[string]interface{} // Can hold various agent types
        Registry *agent.Registry
        Context  context.Context
        Cancel   context.CancelFunc
        Results  map[string]interface{}
        Errors   map[string]error
        mu       sync.RWMutex
}

// NewOrchestrator creates a new orchestrator
func NewOrchestrator(dag *graph.DAG) *Orchestrator {
        ctx, cancel := context.WithCancel(context.Background())
        return &Orchestrator{
                DAG:      dag,
                Agents:   make(map[string]interface{}),
                Registry: agent.NewRegistry(),
                Context:  ctx,
                Cancel:   cancel,
                Results:  make(map[string]interface{}),
                Errors:   make(map[string]error),
        }
}

// RegisterAgent registers an agent with a node
func (o *Orchestrator) RegisterAgent(nodeID string, agentInstance interface{}) error {
        o.mu.Lock()
        defer o.mu.Unlock()

        node, err := o.DAG.GetNode(nodeID)
        if err != nil {
                return fmt.Errorf("node %s not found: %w", nodeID, err)
        }

        o.Agents[nodeID] = agentInstance
        node.SetData(agentInstance)

        return nil
}

// Execute executes the workflow
func (o *Orchestrator) Execute(input interface{}) error {
        // Get topological order
        orderedNodes, err := o.DAG.TopologicalSort()
        if err != nil {
                return fmt.Errorf("failed to get topological order: %w", err)
        }

        log.Printf("Starting workflow execution with %d nodes", len(orderedNodes))

        // Execute nodes in topological order
        for _, node := range orderedNodes {
                log.Printf("Executing node: %s (%s)", node.ID, node.Name)

                // Check if context is cancelled
                if err := o.Context.Err(); err != nil {
                        return fmt.Errorf("workflow cancelled: %w", err)
                }

                // Get agent for this node
                agentInstance, exists := o.Agents[node.ID]
                if !exists {
                        o.mu.Lock()
                        o.Errors[node.ID] = fmt.Errorf("no agent registered for node %s", node.ID)
                        o.mu.Unlock()
                        continue
                }

                // Prepare input for this node
                nodeInput := o.prepareNodeInput(node, input)

                // Execute the agent
                output, err := o.executeAgent(node.ID, agentInstance, nodeInput)
                if err != nil {
                        o.mu.Lock()
                        o.Errors[node.ID] = err
                        node.SetStatus("failed")
                        o.mu.Unlock()
                        log.Printf("Node %s failed: %v", node.ID, err)
                        continue
                }

                // Store result
                o.mu.Lock()
                o.Results[node.ID] = output
                node.SetStatus("completed")
                node.SetData(output)
                o.mu.Unlock()

                log.Printf("Node %s completed successfully", node.ID)
        }

        log.Printf("Workflow execution completed")
        return nil
}

// ExecuteParallel executes the workflow with parallel execution where possible
func (o *Orchestrator) ExecuteParallel(input interface{}) error {
        // Get topological order
        orderedNodes, err := o.DAG.TopologicalSort()
        if err != nil {
                return fmt.Errorf("failed to get topological order: %w", err)
        }

        log.Printf("Starting parallel workflow execution with %d nodes", len(orderedNodes))

        // Track completion status
        completed := make(map[string]bool)
        var mu sync.Mutex

        // Execute nodes in topological order, parallelizing independent nodes
        for _, node := range orderedNodes {
                // Check if context is cancelled
                if err := o.Context.Err(); err != nil {
                        return fmt.Errorf("workflow cancelled: %w", err)
                }

                // Wait for all dependencies to complete
                for {
                        allDepsDone := true
                        deps := o.DAG.GetDependencies(node.ID)
                        mu.Lock()
                        for _, dep := range deps {
                                if !completed[dep.ID] {
                                        allDepsDone = false
                                }
                        }
                        mu.Unlock()

                        if allDepsDone {
                                break
                        }
                        time.Sleep(10 * time.Millisecond)
                }

                // Execute this node in a goroutine
                go func(n *graph.Node) {
                        log.Printf("Executing node: %s (%s)", n.ID, n.Name)

                        // Get agent for this node
                        agentInstance, exists := o.Agents[n.ID]
                        if !exists {
                                o.mu.Lock()
                                o.Errors[n.ID] = fmt.Errorf("no agent registered for node %s", n.ID)
                                o.mu.Unlock()
                                return
                        }

                        // Prepare input for this node
                        nodeInput := o.prepareNodeInput(n, input)

                        // Execute the agent
                        output, err := o.executeAgent(n.ID, agentInstance, nodeInput)
                        if err != nil {
                                o.mu.Lock()
                                o.Errors[n.ID] = err
                                n.SetStatus("failed")
                                mu.Lock()
                                completed[n.ID] = true
                                mu.Unlock()
                                o.mu.Unlock()
                                log.Printf("Node %s failed: %v", n.ID, err)
                                return
                        }

                        // Store result
                        o.mu.Lock()
                        o.Results[n.ID] = output
                        n.SetStatus("completed")
                        n.SetData(output)
                        mu.Lock()
                        completed[n.ID] = true
                        mu.Unlock()
                        o.mu.Unlock()

                        log.Printf("Node %s completed successfully", n.ID)
                }(node)
        }

        // Wait for all nodes to complete
        for {
                allDone := true
                mu.Lock()
                for _, node := range orderedNodes {
                        if !completed[node.ID] {
                                allDone = false
                                break
                        }
                }
                mu.Unlock()

                if allDone {
                        break
                }
                time.Sleep(50 * time.Millisecond)
        }

        log.Printf("Parallel workflow execution completed")
        return nil
}

// prepareNodeInput prepares input for a node based on its dependencies
func (o *Orchestrator) prepareNodeInput(node *graph.Node, initialInput interface{}) interface{} {
        deps := o.DAG.GetDependencies(node.ID)

        if len(deps) == 0 {
                return initialInput
        }

        // If node has dependencies, aggregate their results
        inputs := make([]interface{}, 0, len(deps))
        for _, dep := range deps {
                o.mu.RLock()
                if result, exists := o.Results[dep.ID]; exists {
                        inputs = append(inputs, result)
                }
                o.mu.RUnlock()
        }

        if len(inputs) == 1 {
                return inputs[0]
        }

        return inputs
}

// executeAgent executes an agent (type assertion)
func (o *Orchestrator) executeAgent(nodeID string, agentInstance interface{}, input interface{}) (interface{}, error) {
        // Handle built-in agent types
        switch a := agentInstance.(type) {
        case *agent.DataProcessorAgent:
                return a.Execute(o.Context, input)
        case *agent.TransformAgent:
                return a.Execute(o.Context, input)
        case *agent.ValidationAgent:
                return a.Execute(o.Context, input)
        case *agent.AggregatorAgent:
                return a.Execute(o.Context, input)
        }

        // Handle custom agent types with Execute method
        if executor, ok := agentInstance.(interface {
                Execute(context.Context, interface{}) (interface{}, error)
        }); ok {
                return executor.Execute(o.Context, input)
        }

        return nil, fmt.Errorf("unknown agent type for node %s", nodeID)
}

// GetResult returns the result for a specific node
func (o *Orchestrator) GetResult(nodeID string) (interface{}, error) {
        o.mu.RLock()
        defer o.mu.RUnlock()

        result, exists := o.Results[nodeID]
        if !exists {
                return nil, fmt.Errorf("no result for node %s", nodeID)
        }

        return result, nil
}

// GetError returns the error for a specific node
func (o *Orchestrator) GetError(nodeID string) (error, bool) {
        o.mu.RLock()
        defer o.mu.RUnlock()

        err, exists := o.Errors[nodeID]
        return err, exists
}
